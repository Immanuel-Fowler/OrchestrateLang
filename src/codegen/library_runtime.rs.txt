/// Whether the runtime that starts this library must have Tokio's IO driver: true when a
/// declaration anywhere in the program starts a child process through Tokio (a Python or
/// TypeScript landline, or a secret serverlet). Every runtime needs the time driver.
pub const NEEDS_IO_DRIVER: bool = @NEEDS_IO_DRIVER@;
const ORCH_IO_DRIVER_USERS: &str = @IO_DRIVER_USERS@;

/// Tokio panics, rather than erring, when code needs a driver the runtime was built
/// without. Probing on a named thread turns that into an error before anything is spawned;
/// the host's panic hook still prints the probe's panic under that thread's name.
fn __orch_check_drivers(runtime: &tokio::runtime::Handle) -> Result<(), String> {
    fn probe(runtime: &tokio::runtime::Handle, driver: &str, register: fn()) -> Result<bool, String> {
        let runtime = runtime.clone();
        let thread = std::thread::Builder::new().name(format!("orchestrate {driver} driver check")).spawn(move || {
            let _guard = runtime.enter();
            std::panic::catch_unwind(register).is_ok()
        }).map_err(|e| e.to_string())?;
        Ok(thread.join().unwrap_or(false))
    }
    if !probe(runtime, "time", || {
        let mut sleep = Box::pin(tokio::time::sleep(std::time::Duration::from_secs(1)));
        let _ = std::future::Future::poll(sleep.as_mut(), &mut std::task::Context::from_waker(std::task::Waker::noop()));
    })? {
        return Err("this program needs Tokio's time driver (sleep, landline budgets, and the shutdown grace period use it); build the runtime with enable_time() or enable_all()".into());
    }
    if NEEDS_IO_DRIVER && !probe(runtime, "IO", || {
        #[cfg(unix)]
        let _ = tokio::net::UnixStream::pair();
        #[cfg(not(unix))]
        let _ = std::net::TcpListener::bind("127.0.0.1:0").and_then(tokio::net::TcpListener::from_std);
    })? {
        return Err(format!("this program needs Tokio's IO driver: {ORCH_IO_DRIVER_USERS} start child processes through tokio::process; build the runtime with enable_io() or enable_all()"));
    }
    Ok(())
}

type OrchEvent = std::pin::Pin<Box<dyn std::future::Future<Output = ()> + Send>>;
type OrchEventHandler<T> = std::sync::Arc<dyn Fn(std::sync::Arc<T>) -> OrchEvent + Send + Sync>;

struct OrchContext {

    options: StartOptions,
    events: std::sync::Mutex<std::collections::VecDeque<OrchEvent>>,
    batch: std::sync::Mutex<std::collections::VecDeque<OrchEvent>>,
    clock_nanos: std::sync::atomic::AtomicU64,
    frame: std::sync::atomic::AtomicU64,
    dt_bits: std::sync::atomic::AtomicU64,
    stop_requested: std::sync::atomic::AtomicBool,
    closing: tokio::sync::watch::Sender<bool>,
    shutdown: tokio::sync::watch::Sender<bool>,
    tasks: std::sync::Mutex<Vec<(tokio::task::AbortHandle, bool)>>,
    active: std::sync::atomic::AtomicUsize,
    idle: tokio::sync::Notify,
    assets: std::path::PathBuf,
    /// The runtime the library was started on. Tasks are spawned and timers created
    /// through it, so a tick polled on a thread outside the runtime's context enters nothing.
    runtime: tokio::runtime::Handle,
    /// The program's top-level bindings, once startup has run them.
    program: OrchSlot<__OrchProgram>,
    /// Set whenever an event is queued, so an idle tick skips the queues without locking.
    has_events: std::sync::atomic::AtomicBool,
@EVENT_FIELDS@
}
tokio::task_local! { static ORCH_CONTEXT: std::sync::Arc<OrchContext>; }
thread_local! { static ORCH_SYNC_CONTEXT: std::cell::Cell<*const OrchContext> = const { std::cell::Cell::new(std::ptr::null()) }; }
/// While `tick_sync` polls its body on the calling thread, this names the instance, so the
/// body reaches it through `__orch_context()` without a task-local scope around every tick.
/// A task polled meanwhile — the fallback `block_on` runs other tasks between polls of the
/// body — clears it for its own poll and reaches its instance through the task-local. The
/// previous value comes back when the scope drops, so calls nest.
struct OrchSyncScope(*const OrchContext);
impl OrchSyncScope {
    fn enter(context: &std::sync::Arc<OrchContext>) -> Self {
        Self(ORCH_SYNC_CONTEXT.with(|current| current.replace(std::sync::Arc::as_ptr(context))))
    }
    fn none() -> Self {
        Self(ORCH_SYNC_CONTEXT.with(|current| current.replace(std::ptr::null())))
    }
}
impl Drop for OrchSyncScope {
    fn drop(&mut self) { ORCH_SYNC_CONTEXT.with(|current| current.set(self.0)); }
}
fn __orch_context() -> std::sync::Arc<OrchContext> {
    let sync = ORCH_SYNC_CONTEXT.with(|current| current.get());
    if sync.is_null() { return ORCH_CONTEXT.with(Clone::clone); }
    // SAFETY: `sync` was taken with `Arc::as_ptr` from the Arc that a `tick_sync` call
    // borrows from its `Scripts` for the whole call, and the scope that set it drops before
    // that call returns, so the allocation is alive: this is `Arc::clone` through a raw pointer.
    unsafe { std::sync::Arc::increment_strong_count(sync); std::sync::Arc::from_raw(sync) }
}
/// A task of this instance: the task-local names the instance, and the sync pointer is
/// cleared around each poll, since a sync tick's `block_on` polls tasks in between.
fn __orch_task<F>(context: std::sync::Arc<OrchContext>, future: F) -> impl std::future::Future<Output = F::Output> + Send
where F: std::future::Future + Send + 'static, F::Output: Send + 'static {
    ORCH_CONTEXT.scope(context, async move {
        let mut future = std::pin::pin!(future);
        std::future::poll_fn(|cx| { let _task = OrchSyncScope::none(); future.as_mut().poll(cx) }).await
    })
}
fn __orch_log(level: LogLevel, message: String) { __orch_context().host.log(level, &message); }
fn stop_orch() { __orch_context().stop_requested.store(true, std::sync::atomic::Ordering::Release); }
/// Out of line, so a host call in a hook is the trait call and one branch. A failed call is
/// logged through the host and yields the return type's default, as it always has.
#[cold]
#[inline(never)]
fn __orch_host_failed(name: &str, error: &str) {
    __orch_log(LogLevel::Error, format!("[orchestrate] host call {name} failed: {error}"));
}
/// `tokio::time::timeout` for code whose first poll may happen on a thread outside the
/// runtime's context, as a synchronous tick's does: the timer is created under the runtime.
fn __orch_timeout<F: std::future::Future>(duration: std::time::Duration, future: F) -> tokio::time::Timeout<F> {
    let context = __orch_context();
    let _runtime = context.runtime.enter();
    tokio::time::timeout(duration, future)
}

/// The program's state between ticks, entered by a synchronous tick on the calling thread
/// or by the coordinator on the runtime. Each side raises its own flag and then reads the
/// other's, all sequentially consistent, so the two cannot both find the other's flag
/// clear. A synchronous tick that finds the coordinator's flag raised fails with the
/// "already running" error; a coordinator that finds the tick's raised registers with
/// `released` and waits. So the fast path is two stores and two loads, with no
/// read-modify-write, and it neither waits nor notifies.
///
/// Invariant: `value` is reached only through a guard, and a guard is created only by a
/// side that raised its own flag and then observed the other side's flag clear. Both
/// observing that is impossible (a store-buffering outcome, which sequential consistency
/// rules out), each side is one caller at a time (`tick_sync` takes `&mut Scripts`; the
/// coordinator is one task), so two guards never exist at once and the value is never
/// aliased. That is a mutex's guarantee, which is why the slot is Sync when its value is
/// Send, and why the guard, like a mutex guard, is Send but not Sync.
struct OrchSlot<T> {
    /// Raised by `tick_sync` while it holds the slot.
    sync_held: std::sync::atomic::AtomicBool,
    /// Raised by the coordinator while it holds the slot.
    task_held: std::sync::atomic::AtomicBool,
    /// Set by a coordinator that found the slot held; the tick that releases it notifies.
    contended: std::sync::atomic::AtomicBool,
    released: tokio::sync::Notify,
    value: std::cell::UnsafeCell<Option<T>>,
}
unsafe impl<T: Send> Sync for OrchSlot<T> {}
struct OrchSlotGuard<'a, T> {
    slot: &'a OrchSlot<T>,
    sync: bool,
    _not_sync: std::marker::PhantomData<std::cell::Cell<()>>,
}
impl<T> OrchSlot<T> {
    fn new() -> Self {
        Self {
            sync_held: false.into(), task_held: false.into(), contended: false.into(),
            released: tokio::sync::Notify::new(), value: std::cell::UnsafeCell::new(None),
        }
    }
    /// From the calling thread: fails rather than waits when the coordinator holds the slot.
    fn try_enter_sync(&self) -> Option<OrchSlotGuard<'_, T>> {
        self.sync_held.store(true, std::sync::atomic::Ordering::SeqCst);
        if self.task_held.load(std::sync::atomic::Ordering::SeqCst) {
            self.release_sync();
            return None;
        }
        Some(OrchSlotGuard { slot: self, sync: true, _not_sync: std::marker::PhantomData })
    }
    /// From the coordinator: waits while a synchronous tick holds the slot.
    async fn enter(&self) -> OrchSlotGuard<'_, T> {
        self.task_held.store(true, std::sync::atomic::Ordering::SeqCst);
        loop {
            if !self.sync_held.load(std::sync::atomic::Ordering::SeqCst) { break; }
            // Registered before the flag and the retry: notify_waiters reaches a Notified
            // created before it, so a release between the retry and the wait is not lost.
            let released = self.released.notified();
            self.contended.store(true, std::sync::atomic::Ordering::SeqCst);
            if !self.sync_held.load(std::sync::atomic::Ordering::SeqCst) { break; }
            released.await;
        }
        OrchSlotGuard { slot: self, sync: false, _not_sync: std::marker::PhantomData }
    }
    fn release_sync(&self) {
        self.sync_held.store(false, std::sync::atomic::Ordering::SeqCst);
        // A load on the fast path. The coordinator stores `contended` and then loads
        // `sync_held`; this side stores `sync_held` and then loads `contended`; both
        // sequentially consistent, so a registration and this release cannot both go unseen.
        if self.contended.load(std::sync::atomic::Ordering::SeqCst) {
            self.contended.store(false, std::sync::atomic::Ordering::SeqCst);
            self.released.notify_waiters();
        }
    }
}
impl<T> std::ops::Deref for OrchSlotGuard<'_, T> {
    type Target = Option<T>;
    // SAFETY: the guard is the only path to the value, and only one exists; see OrchSlot.
    fn deref(&self) -> &Option<T> { unsafe { &*self.slot.value.get() } }
}
impl<T> std::ops::DerefMut for OrchSlotGuard<'_, T> {
    // SAFETY: as for deref, through the guard's exclusive borrow.
    fn deref_mut(&mut self) -> &mut Option<T> { unsafe { &mut *self.slot.value.get() } }
}
impl<T> Drop for OrchSlotGuard<'_, T> {
    fn drop(&mut self) {
        if self.sync { self.slot.release_sync(); } else { self.slot.task_held.store(false, std::sync::atomic::Ordering::SeqCst); }
    }
}

struct OrchTaskGuard(std::sync::Arc<OrchContext>);
impl Drop for OrchTaskGuard {
    fn drop(&mut self) {
        self.0.active.fetch_sub(1, std::sync::atomic::Ordering::AcqRel);
        self.0.idle.notify_one();
    }
}
fn __orch_spawn<F>(future: F) -> tokio::task::JoinHandle<F::Output>
where F: std::future::Future + Send + 'static, F::Output: Send + 'static {
    __orch_spawn_kind(future, false)
}
fn __orch_spawn_line<F>(future: F) -> tokio::task::JoinHandle<F::Output>
where F: std::future::Future + Send + 'static, F::Output: Send + 'static {
    __orch_spawn_kind(future, true)
}
fn __orch_spawn_kind<F>(future: F, line: bool) -> tokio::task::JoinHandle<F::Output>
where F: std::future::Future + Send + 'static, F::Output: Send + 'static {
    let context = __orch_context();
    assert!(!context.options.deterministic, "deterministic mode supports lifecycle and event hooks; spawned workers and serverlets are excluded");
    context.active.fetch_add(1, std::sync::atomic::Ordering::AcqRel);
    let guard = OrchTaskGuard(context.clone());
    let task = context.runtime.spawn(__orch_task(context.clone(), async move {
        let _guard = guard;
        future.await
    }));
    let mut tasks = context.tasks.lock().unwrap();
    tasks.retain(|(task, _)| !task.is_finished());
    if *context.closing.borrow() { task.abort(); }
    tasks.push((task.abort_handle(), line));
    task
}
async fn __orch_recv<T>(rx: &mut tokio::sync::mpsc::Receiver<T>) -> Option<T> {
    let context = __orch_context();
    let mut closing = context.closing.subscribe();
    if *closing.borrow() { return None; }
    tokio::select! { biased; _ = closing.changed() => None, value = rx.recv() => value }
}
async fn __orch_read_frame<R: tokio::io::AsyncReadExt + Unpin>(reader: &mut R) -> std::io::Result<Option<OrchFrame>> {
    let context = __orch_context();
    let mut closing = context.closing.subscribe();
    if *closing.borrow() { return Ok(None); }
    tokio::select! { biased; _ = closing.changed() => Ok(None), result = __secret_read_frame(reader) => result }
}
async fn __orch_sleep(duration: std::time::Duration) {
    let context = __orch_context();
    if !context.options.deterministic {
        // Created under the runtime, since a synchronous tick polls its body outside it.
        let sleep = { let _runtime = context.runtime.enter(); tokio::time::sleep(duration) };
        sleep.await;
        return;
    }
    assert!(ORCH_EVENT.try_with(|_| ()).is_ok(), "deterministic sleep is supported only in event handlers");
    let deadline = context.clock_nanos.load(std::sync::atomic::Ordering::Relaxed).saturating_add(duration.as_nanos().min(u64::MAX as u128) as u64);
    std::future::poll_fn(|_| if context.clock_nanos.load(std::sync::atomic::Ordering::Relaxed) >= deadline { std::task::Poll::Ready(()) } else { std::task::Poll::Pending }).await
}
impl OrchContext {
    fn advance_time(&self, dt: f64) {
        assert!(dt.is_finite() && dt >= 0.0, "tick dt must be finite and nonnegative");
        // Whole nanoseconds, rounded as Duration::from_secs_f64 rounds them (ties to even)
        // and saturating where it would panic. Only a tick writes the clock, and it holds the
        // program slot while it does, so a load and a store replace the atomic add.
        let nanos = (dt * 1_000_000_000.0).round_ties_even() as u64;
        self.clock_nanos.store(self.clock_nanos.load(std::sync::atomic::Ordering::Relaxed).wrapping_add(nanos), std::sync::atomic::Ordering::Relaxed);
    }
    /// Handle the events queued when the drain starts, and only those: a handler that
    /// triggers an event leaves it in `events` for the next drain, so a handler that
    /// re-triggers itself cannot keep one tick running forever. `batch` also carries the
    /// fan-out from a queued event to its handlers, which belongs to this drain.
    fn queue_event(&self, event: OrchEvent) {
        self.events.lock().unwrap().push_back(event);
        self.has_events.store(true, std::sync::atomic::Ordering::Release);
    }
    /// Whether a drain has anything to do: checked before `drain_events` is even created,
    /// since building and polling that future costs more than the idle tick around it.
    #[inline]
    fn events_pending(&self) -> bool { self.has_events.load(std::sync::atomic::Ordering::Acquire) }
    async fn drain_events(&self) {
        // Cleared before the queue is read, so an event queued during the drain is seen next time.
        if !self.has_events.swap(false, std::sync::atomic::Ordering::AcqRel) { return; }
        {
            let mut batch = self.batch.lock().unwrap();
            batch.append(&mut self.events.lock().unwrap());
        }
        let mut pending = std::collections::VecDeque::new();
        loop {
            let event = self.batch.lock().unwrap().pop_front();
            let Some(event) = event else { break; };
            if self.options.deterministic {
                let mut event = event;
                let result = ORCH_EVENT.scope((), std::future::poll_fn(|cx| std::task::Poll::Ready(event.as_mut().poll(cx)))).await;
                if result.is_pending() { pending.push_back(event); }
            } else { event.await; }
        }
        if !pending.is_empty() {
            self.events.lock().unwrap().extend(pending);
            self.has_events.store(true, std::sync::atomic::Ordering::Release);
        }
    }
    async fn cleanup(&self) {
        self.closing.send_replace(true);
        for (task, line) in self.tasks.lock().unwrap().iter() { if !line { task.abort(); } }
        let drained = async {
            loop {
                let notified = self.idle.notified();
                if self.active.load(std::sync::atomic::Ordering::Acquire) == 0 { break; }
                notified.await;
            }
        };
        if tokio::time::timeout(self.options.shutdown_grace, drained).await.is_err() {
            for (task, _) in self.tasks.lock().unwrap().iter() { task.abort(); }
            while self.active.load(std::sync::atomic::Ordering::Acquire) != 0 { self.idle.notified().await; }
        }
        let _ = std::fs::remove_dir_all(&self.assets);
    }
}

enum OrchCommand {
    Tick(f64@TICK_FIELD@, tokio::sync::oneshot::Sender<@TICK_OUTPUT@>),
    FixedTick(f64, tokio::sync::oneshot::Sender<()>),
    Shutdown,
}
/// A tick polled on the calling thread: the instance is reachable through the thread for
/// every poll of the body, the first one and those under `block_on` alike.
struct OrchSyncTick<'a, F> {
    context: &'a std::sync::Arc<OrchContext>,
    future: std::pin::Pin<&'a mut F>,
}
impl<F: std::future::Future> std::future::Future for OrchSyncTick<'_, F> {
    type Output = F::Output;
    fn poll(mut self: std::pin::Pin<&mut Self>, cx: &mut std::task::Context<'_>) -> std::task::Poll<F::Output> {
        let _scope = OrchSyncScope::enter(self.context);
        self.future.as_mut().poll(cx)
    }
}
pub struct Scripts {
    commands: tokio::sync::mpsc::UnboundedSender<OrchCommand>,
    ready: Option<tokio::sync::oneshot::Receiver<()>>,
    supervisor: Option<tokio::task::JoinHandle<Result<(), String>>>,
    context: std::sync::Arc<OrchContext>,
}
impl Scripts {
    /// Pump a current-thread runtime until startup is complete. No work runs between pumps.
    pub fn ready_blocking(&mut self, runtime: &tokio::runtime::Runtime) -> Result<(), String> {
        runtime.block_on(self.ready())
    }
    pub fn tick_blocking(&mut self, runtime: &tokio::runtime::Runtime, dt: f64@TICK_PARAM@) -> Result<@TICK_OUTPUT@, String> {
        runtime.block_on(self.tick(dt@TICK_ARG@))
    }
    pub fn shutdown_blocking(self, runtime: &tokio::runtime::Runtime) -> Result<(), String> {
        runtime.block_on(self.shutdown())
    }

    /// Wait until top-level state and on_start have initialized.
    pub async fn ready(&mut self) -> Result<(), String> {
        if let Some(ready) = self.ready.take() {
            ready.await.map_err(|_| "library startup task failed".to_string())?;
        }
        Ok(())
    }
    /// Run every on_tick hook once, in declaration order. Bound slow landline calls made
    /// inside hooks with a landline `budget`.
    pub async fn tick(&mut self, dt: f64@TICK_PARAM@) -> Result<@TICK_OUTPUT@, String> {
        self.ready().await?;
        if self.stop_requested() { return Err("orchestration requested stop".into()); }
        let (tx, rx) = tokio::sync::oneshot::channel();
        self.commands.send(OrchCommand::Tick(dt@TICK_ARG@, tx)).map_err(|_| "library is stopped")?;
        rx.await.map_err(|_| "tick task failed".into())
    }
    pub async fn fixed_tick(&mut self, step: f64) -> Result<(), String> {
        self.ready().await?;
        if self.stop_requested() { return Err("orchestration requested stop".into()); }
        let (tx, rx) = tokio::sync::oneshot::channel();
        self.commands.send(OrchCommand::FixedTick(step, tx)).map_err(|_| "library is stopped")?;
        rx.await.map_err(|_| "fixed tick task failed".into())
    }
    pub fn fixed_tick_blocking(&mut self, runtime: &tokio::runtime::Runtime, step: f64) -> Result<(), String> {
        runtime.block_on(self.fixed_tick(step))
    }

    /// Run the tick on the calling thread, with no coordinator task, command channel, park,
    /// or entry into the runtime: a body that finishes without waiting returns straight
    /// away. One that has to wait finishes under `runtime.block_on`, so the result matches
    /// `tick_blocking`. The `shutdown` flag needs no check here: both ways of setting it
    /// consume the `Scripts` this call borrows.
    pub fn tick_sync(&mut self, runtime: &tokio::runtime::Runtime, dt: f64@TICK_PARAM@) -> Result<@TICK_OUTPUT@, String> {
        if self.ready.is_some() { self.ready_blocking(runtime)?; }
        if self.stop_requested() { return Err("orchestration requested stop".into()); }
        let context = &self.context;
        let mut state = context.program.try_enter_sync().ok_or("a tick is already running")?;
        let program = state.as_mut().ok_or("library is stopped")?;
        // Only a tick writes these, and it holds the program slot, so plain loads and stores do.
        context.frame.store(context.frame.load(std::sync::atomic::Ordering::Relaxed).wrapping_add(1), std::sync::atomic::Ordering::Relaxed);
        context.dt_bits.store(dt.to_bits(), std::sync::atomic::Ordering::Relaxed);
        let mut body = std::pin::pin!(__orch_run_tick(context, program, dt@TICK_ARG@));
        let mut tick = OrchSyncTick { context, future: body.as_mut() };
        match std::future::Future::poll(std::pin::Pin::new(&mut tick), &mut std::task::Context::from_waker(std::task::Waker::noop())) {
            std::task::Poll::Ready(output) => Ok(output),
            std::task::Poll::Pending => Ok(runtime.block_on(tick)),
        }
    }
    /// `fixed_tick` on the calling thread, the way `tick_sync` runs `tick`.
    pub fn fixed_tick_sync(&mut self, runtime: &tokio::runtime::Runtime, step: f64) -> Result<(), String> {
        if self.ready.is_some() { self.ready_blocking(runtime)?; }
        if self.stop_requested() { return Err("orchestration requested stop".into()); }
        let context = &self.context;
        let mut state = context.program.try_enter_sync().ok_or("a tick is already running")?;
        let program = state.as_mut().ok_or("library is stopped")?;
        let mut body = std::pin::pin!(__orch_run_fixed_tick(context, program, step));
        let mut tick = OrchSyncTick { context, future: body.as_mut() };
        match std::future::Future::poll(std::pin::Pin::new(&mut tick), &mut std::task::Context::from_waker(std::task::Waker::noop())) {
            std::task::Poll::Ready(()) => Ok(()),
            std::task::Poll::Pending => Ok(runtime.block_on(tick)),
        }
    }
    pub fn stop_requested(&self) -> bool {
        self.context.stop_requested.load(std::sync::atomic::Ordering::Acquire)
    }
    /// Run on_stop, close sidecars, and await owned task cleanup. Does not stop Tokio.
    pub async fn shutdown(mut self) -> Result<(), String> {
        self.context.shutdown.send_replace(true);
        let _ = self.commands.send(OrchCommand::Shutdown);
        self.supervisor.take().unwrap().await.map_err(|e| e.to_string())?
    }
}
impl Drop for Scripts {
    fn drop(&mut self) { self.context.shutdown.send_replace(true); let _ = self.commands.send(OrchCommand::Shutdown); }
}

pub fn start(runtime: &tokio::runtime::Handle, host: impl Host) -> Result<Scripts, String> {
    start_with_options(runtime, host, StartOptions::default())
}
pub fn start_with_options(runtime: &tokio::runtime::Handle, host: impl Host, options: StartOptions) -> Result<Scripts, String> {
    __orch_check_drivers(runtime)?;
    static NEXT_INSTANCE: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
    let assets = loop {
        let id = NEXT_INSTANCE.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
        let path = std::env::temp_dir().join(format!("orch_library_{}_{}", std::process::id(), id));
        match std::fs::create_dir(&path) {
            Ok(()) => break path,
            Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => continue,
            Err(e) => return Err(e.to_string()),
        }
    };
    for (name, contents) in ORCH_ASSETS {
        let path = assets.join(name);
        let result = std::fs::create_dir_all(path.parent().unwrap()).and_then(|_| std::fs::write(&path, contents));
        if let Err(e) = result { let _ = std::fs::remove_dir_all(&assets); return Err(e.to_string()); }
        #[cfg(unix)]
        if name.starts_with("secret_") || name.ends_with("/serverlet") || name.ends_with("/serverlet.exe") {
            use std::os::unix::fs::PermissionsExt;
            if let Err(e) = std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o700)) {
                let _ = std::fs::remove_dir_all(&assets); return Err(e.to_string());
            }
        }
    }
    let (closing, _) = tokio::sync::watch::channel(false);
    let (shutdown, _) = tokio::sync::watch::channel(false);
    let context = std::sync::Arc::new(OrchContext {
        host: std::sync::Arc::new(host), options, events: Default::default(), batch: Default::default(), clock_nanos: 0.into(), frame: 0.into(), dt_bits: 0.into(), stop_requested: false.into(), closing, shutdown,
        tasks: Default::default(), active: 0.into(), idle: tokio::sync::Notify::new(), assets, runtime: runtime.clone(),
        program: OrchSlot::new(), has_events: false.into(),
@EVENT_INIT@
    });
    let (commands, rx) = tokio::sync::mpsc::unbounded_channel();
    let (ready_tx, ready) = tokio::sync::oneshot::channel();
    let coordinator = runtime.spawn(__orch_task(context.clone(), __orch_entry(rx, ready_tx)));
    let cleanup_context = context.clone();
    let supervisor = runtime.spawn(async move {
        let result = coordinator.await.map_err(|e| e.to_string());
        cleanup_context.cleanup().await;
        result
    });
    Ok(Scripts { commands, ready: Some(ready), supervisor: Some(supervisor), context })
}
