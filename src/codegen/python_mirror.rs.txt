/// `start @NAME@ from path`: the same landline, running the source at `path` rather than the
/// one it was built with. The file has to be there; whether it implements the declaration
/// is found in the handshake, as for any start.
#[allow(non_snake_case)]
pub fn start_@NAME@_from(path: String) -> Result<@NAME@Client, String> {
    let source = std::fs::canonicalize(&path)
        .ok()
        .filter(|source| source.is_file())
        .ok_or_else(|| format!("landline '@NAME@': no file '{}'", path))?;
    Ok(__start_@NAME@_with(Some(source.to_string_lossy().to_string())))
}
