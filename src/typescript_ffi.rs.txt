use std::io::{Read, Write};
struct Child(std::process::Child);
impl Drop for Child { fn drop(&mut self) { let _ = self.0.kill(); let _ = self.0.wait(); } }
fn read_frame(reader: &mut impl Read) -> OrchFrame {
    let mut size = [0;4]; reader.read_exact(&mut size).expect("TypeScript FFI closed its pipe");
    let n = u32::from_le_bytes(size) as usize;
    assert!((5..=64*1024*1024).contains(&n), "invalid TypeScript FFI frame length");
    let mut bytes = vec![0;n]; reader.read_exact(&mut bytes).expect("truncated TypeScript FFI frame");
    OrchFrame::parse(bytes).expect("invalid frame")
}
fn write_frame(writer: &mut impl Write, kind: u8, id: u32, payload: &[u8]) {
    writer.write_all(&((payload.len()+5) as u32).to_le_bytes()).unwrap();
    writer.write_all(&[kind]).unwrap(); writer.write_all(&id.to_le_bytes()).unwrap();
    writer.write_all(payload).unwrap(); writer.flush().unwrap();
}

/// One TypeScript process, kept alive across calls. Starting it costs a process
/// spawn and a handshake, so a per-call process would put both on every call.
struct Bridge {
    _child: Child,
    input: std::process::ChildStdin,
    output: std::process::ChildStdout,
    next_call_id: u32,
}

impl Bridge {
    fn start(executable: &std::path::Path, signatures: &[String]) -> Bridge {
        let mut child = Child(std::process::Command::new(executable)
            .stdin(std::process::Stdio::piped())
            .stdout(std::process::Stdio::piped())
            .spawn()
            .expect("cannot start TypeScript FFI"));
        let mut input = child.0.stdin.take().unwrap();
        let mut output = child.0.stdout.take().unwrap();
        let hello = read_frame(&mut output);
        assert_eq!((hello.kind, hello.call_id), (1, 0), "expected HELLO");
        let mut pos = 0;
        assert_eq!(i64::wire_decode(&hello.payload, &mut pos), Some(1));
        assert_eq!(Vec::<String>::wire_decode(&hello.payload, &mut pos), Some(signatures.to_vec()));
        assert_eq!(pos, hello.payload.len());
        write_frame(&mut input, 2, 0, &[]);
        Bridge { _child: child, input, output, next_call_id: 1 }
    }

    /// A handler error arrives as a complete ERROR frame, so the pipe is still in
    /// sync and the process keeps its state; only a broken frame panics.
    fn invoke(&mut self, payload: &[u8]) -> Result<Vec<u8>, String> {
        let id = self.next_call_id;
        self.next_call_id = self.next_call_id.wrapping_add(1).max(1);
        write_frame(&mut self.input, 3, id, payload);
        let reply = read_frame(&mut self.output);
        assert_eq!(reply.call_id, id, "TypeScript FFI replied to the wrong call");
        if reply.kind == 5 {
            return Err(String::wire_decode(&reply.payload, &mut 0).unwrap_or_default());
        }
        assert_eq!(reply.kind, 4, "expected REPLY");
        Ok(reply.payload)
    }
}

static BRIDGE: std::sync::OnceLock<std::sync::Mutex<Option<Bridge>>> = std::sync::OnceLock::new();

fn call(executable: std::path::PathBuf, payload: Vec<u8>, signatures: Vec<String>) -> Vec<u8> {
    let slot = BRIDGE.get_or_init(|| std::sync::Mutex::new(None));
    let mut guard = slot.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
    // Taken for the duration of the call: a panic mid-frame leaves the slot empty
    // rather than leaving a desynchronised pipe for the next caller to reuse.
    let mut bridge = guard.take().unwrap_or_else(|| Bridge::start(&executable, &signatures));
    let reply = bridge.invoke(&payload);
    *guard = Some(bridge);
    match reply {
        Ok(payload) => payload,
        Err(message) => panic!("TypeScript FFI: {message}"),
    }
}
